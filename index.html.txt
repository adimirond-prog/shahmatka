    function renderTable() {

      const query = document.getElementById("searchBox").value.toLowerCase().trim();

                 it.curator.toLowerCase().includes(query);

`;
          tr.appendChild(tdObj);

          // Куратор (объединен, если одинаковый)
          if (!allSameCurator || idx === 0) {
            const tdCur = document.createElement("td");
            if (allSameCurator) tdCur.rowSpan = filteredItems.length;

            if (isEditor) {
              tdCur.innerHTML = `
                <textarea style="width:100%; background:#0f141c; border:1px solid var(--gold-dark); color:#fff; border-radius:6px; padding:6px; font-size:0.85rem;" onchange="updateCurator('${currentTab}', '${group.dev}', ${idx}, this.value)">${item.curator}</textarea>
              `;
            } else {
              tdCur.innerHTML = `
                <div class="curator-box">
                  <b>${item.curator.replace('\n', '<br>')}</b>
                </div>
              `;
            }
            tr.appendChild(tdCur);
          }

          // Фиксация (объединена на всю группу)
          if (idx === 0) {
            const tdFix = document.createElement("td");
            tdFix.rowSpan = filteredItems.length;
            let btnHtml = group.fixUrl ? `<div style="margin-top:6px;"><a href="${group.fixUrl}" target="_blank" class="btn-fix">Зафиксировать клиента ➔</a></div>` : '';
            tdFix.innerHTML = `
              <div class="rules-text">${group.fixation}</div>
              ${btnHtml}
            `;
            tr.appendChild(tdFix);
          }

          tbody.appendChild(tr);
        });
      });
    }

    function loginEditor() {
      const val = document.getElementById("authInput").value.trim().toLowerCase();
      if (val === 'admin2026' || allowedEmails.includes(val)) {
        isEditor = true;
        document.getElementById("roleText").textContent = "Режим: Администратор / Редактор";
        document.getElementById("roleText").style.color = "var(--gold)";
        document.getElementById("btnAdminAccess").textContent = "Выйти из режима";
        document.getElementById("btnAdminAccess").onclick = logoutEditor;
        document.getElementById("btnManageEmails").style.display = "inline-block";
        closeAuthModal();
        renderTable();
        alert("Доступ открыт! Теперь вы можете редактировать данные кураторов прямо на сайте.");
      } else {
        alert("Email или пароль не найден в списке доверенных редакторов!");
      }
    }

    function logoutEditor() {
      isEditor = false;
      document.getElementById("roleText").textContent = "Режим: Просмотр (для сотрудников)";
      document.getElementById("roleText").style.color = "var(--text-main)";
      document.getElementById("btnAdminAccess").textContent = "Вход для редактора";
      document.getElementById("btnAdminAccess").onclick = openAuthModal;
      document.getElementById("btnManageEmails").style.display = "none";
      renderTable();
    }

    // Управление email редакторов
    function openEmailsModal() { document.getElementById("emailsModal").classList.add("open"); }
    function closeEmailsModal() { document.getElementById("emailsModal").classList.remove("open"); }

    function renderEmailsList() {
      const container = document.getElementById("emailsList");
      container.innerHTML = "";
      allowedEmails.forEach((email, i) => {
        const item = document.createElement("div");
        item.className = "email-chip";
        item.innerHTML = `<span>${email}</span><span class="btn-del" onclick="deleteEmail(${i})">✕</span>`;
        container.appendChild(item);
      });
    }

    function addEditorEmail() {
      const inp = document.getElementById("newEmailInput");
      const val = inp.value.trim().toLowerCase();
      if (val && !allowedEmails.includes(val)) {
        allowedEmails.push(val);
localStorage.setItem('allowed_editors', JSON.stringify(allowedEmails));
        inp.value = "";
        renderEmailsList();
      }
    }

    function deleteEmail(idx) {
      allowedEmails.splice(idx, 1);
      localStorage.setItem('allowed_editors', JSON.stringify(allowedEmails));
      renderEmailsList();
    }

    function updateCurator(tab, dev, idx, newVal) {
      const grp = database[tab].find(g => g.dev === dev);
      if (grp) {
        grp.items[idx].curator = newVal;
      }
    }
